from flask import Flask, render_template, request, redirect, url_for, flash, session app = Flask(__name__) app.secret_key = "smartcart_secure_key" # Mock item dictionary representing your database products_db = { 1: {"product_id": 1, "name": "Wireless Headphones", "category": "Electronics", "price": 4999.00, "image": "headphone.jpg", "description": "Noise-cancelling"} } # --- UNIFIED CONTROLLERS FOR PRODUCT MODIFICATION --- # Pattern 1: Catching updates from /admin/update-item/ @app.route('/admin/update-item/', methods=['GET', 'POST']) def update_item(product_id): if 'admin_user' not in session: return redirect(url_for('admin_login')) product = products_db.get(product_id) if not product: flash("Target product not found in inventory catalog.", "danger") return redirect(url_for('item_list')) if request.method == 'POST': # Apply structural mutations cleanly from field names product['name'] = request.form.get('name') product['category'] = request.form.get('category') product['price'] = float(request.form.get('price', 0.00)) product['description'] = request.form.get('description', '') if request.form.get('image'): product['image'] = request.form.get('image') flash(f"Product #{product_id} specs successfully synchronized.", "success") # Reloading to the same route prevents browser resubmission warnings return redirect(url_for('update_item', product_id=product_id)) return render_template('edit_item.html', product=product) # Pattern 2: Alias catch point for /admin/edit-item/ to prevent routing drift broken link errors @app.route('/admin/edit-item/', methods=['GET', 'POST']) def edit_item_alias(product_id): # Seamlessly hands off payload management straight to our primary handler block return update_item(product_id) @app.route('/admin/item-list') def item_list(): return render_template('item_list.html', products=list(products_db.values()))
